function clean($str) {
    $str = strtolower(trim($str));
    $str = preg_replace('/[^a-z0-9_\-]/', '_', $str);
    return preg_replace('/_+/', '_', $str);
}

$message = "";

if ($_server['request_method'] === 'post') {

    $mode = $_post['mode'] ?? 'legaisee';
    $title = trim($_post['title'] ?? 'untitled');
    $source = trim($_post['source'] ?? 'manual');
    $content = "";

    // handle file upload
    if (!empty($_files['file']['tmp_name'])) {
        $content = file_get_contents($_files['file']['tmp_name']);
    }

    // handle direct paste
    if (!empty($_post['paste'])) {
        $content = $_post['paste'];
    }

    if (trim($content) === "") {
        $message = "<div class='error'>no content provided</div>";
    } else {

        $timestamp = date("ymd_his");
        $safetitle = clean($title);

        // 🔹 legaisee mode
        if ($mode === "legaisee") {

            $client = clean($_post['client'] ?? 'general');
            $project = clean($_post['project'] ?? 'default');
            $session = clean($_post['session'] ?? 'session');

            $basedir = $_server['document_root'] . "/commandcenter/clients/";
            $targetdir = $basedir . "$client/$project/$session/raw/";

            if (!is_dir($targetdir)) {
                mkdir($targetdir, 0755, true);
            }

            $filename = $timestamp . "_" . $safetitle . ".md";
            $fullpath = $targetdir . $filename;

            file_put_contents($fullpath, $content);

            $meta = [
                "client" => $client,
                "project" => $project,
                "session" => $session,
                "source" => $source,
                "title" => $title,
                "entry_type" => "raw",
                "timestamp" => date("y-m-d h:i:s")
            ];

            file_put_contents($fullpath . ".json", json_encode($meta, json_pretty_print));

        }

        // 🔹 general storage mode
        else {

            $basedir = $_server['document_root'] . "/commandcenter/library/";
            if (!is_dir($basedir)) {
                mkdir($basedir, 0755, true);
            }

            $filename = $timestamp . "_" . $safetitle . ".md";
            $fullpath = $basedir . $filename;

            file_put_contents($fullpath, $content);

            $meta = [
                "source" => $source,
                "title" => $title,
                "entry_type" => "library",
                "timestamp" => date("y-m-d h:i:s")
            ];

            file_put_contents($fullpath . ".json", json_encode($meta, json_pretty_print));
        }

        // 🔹 normalized copy (for search)
        $normalizeddir = __dir__ . "/normalized/";
        if (!is_dir($normalizeddir)) {
            mkdir($normalizeddir, 0755, true);
        }

        $normfile = $normalizeddir . $filename;
        file_put_contents($normfile, strtolower($content));
        file_put_contents($normfile . ".json", json_encode($meta, json_pretty_print));

        $message = "<div class='success'>✓ ingested successfully</div>";
    }
}
